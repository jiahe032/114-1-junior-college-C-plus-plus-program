#include <stdio.h>

float BMI(float height, float weight);

int main()
{
    float height, weight, bmi;

    printf("請輸入身高（公分）：");
    scanf_s("%f", &height);

    printf("請輸入體重（公斤）：");
    scanf_s("%f", &weight);

    bmi = BMI(height, weight);

    printf("您的BMI為：%.2f\n", bmi);

    return 0;
}

float BMI(float height, float weight)
{
    height = height / 100; // cm 轉 m
    return weight / (height * height);
}
